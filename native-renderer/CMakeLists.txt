# XDroid360's native (EDRAM-free) Vulkan renderer, from nfsmw-android (NOTICE.md). A static
# library the game runtime links next to Xenia's Vulkan GPU emulation; the runtime picks one at
# startup ([video] gpu_renderer in config.toml).
#
# XDROID360_DEPS_ROOT: a checkout with the recomp submodule (Xenia's base and GPU headers,
# xenia-canary's third-party sources). XDROID360_SOURCE_ROOT: this repository (xenia-vulkan).
cmake_minimum_required(VERSION 3.22)
project(XDroid360NativeRenderer C CXX)

# C++23: the renderer uses std::byteswap.
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)

if(NOT XDROID360_SOURCE_ROOT)
    get_filename_component(XDROID360_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
endif()
if(NOT XDROID360_DEPS_ROOT)
    set(XDROID360_DEPS_ROOT "${XDROID360_SOURCE_ROOT}")
endif()
set(XE_CANARY "${XDROID360_DEPS_ROOT}/recomp/tools/xenia-canary")

# All of the renderer's Vulkan files, on the compat layer.
set(NATIVE_RENDERER_SOURCES
    compat/rex_cvar.cpp
    compat/rex_filesystem.cpp
    compat/rex_host.cpp
    compat/xdroid360_game_links.cpp
    src/nfsmw_nativo_sistema.cpp
    src/nfsmw_ajustes_graficos.cpp
    src/nfsmw_shader_library.cpp
    src/nfsmw_spirv_precision.cpp
    src/nfsmw_nativo_texturas_pool.cpp
    src/nfsmw_nativo_shaders.cpp
    src/nfsmw_nativo_captura.cpp
    src/nfsmw_nativo_destinos.cpp
    src/nfsmw_nativo_dibujos.cpp
    src/nfsmw_nativo_ganchos.cpp
    src/nfsmw_espera_anillo.cpp)

add_library(native-renderer STATIC ${NATIVE_RENDERER_SOURCES})
target_include_directories(native-renderer PRIVATE
    compat
    src
    "${XDROID360_SOURCE_ROOT}/xenia-vulkan/src"
    "${XDROID360_SOURCE_ROOT}/xenia-vulkan/src/xenia/ui/shaders/bytecode"
    "${XDROID360_DEPS_ROOT}/recomp/xenia"
    "${XDROID360_DEPS_ROOT}/recomp/xenia/src"
    "${XE_CANARY}/third_party/fmt/include"
    "${XE_CANARY}/third_party/xxhash")
# Xenia's presenter shaders as SPIR-V headers, generated by the game runtime's build.
if(XDROID360_GEN_DIR)
    target_include_directories(native-renderer PRIVATE "${XDROID360_GEN_DIR}/xenia/ui/shaders/bytecode")
endif()
target_include_directories(native-renderer SYSTEM PRIVATE
    "${XE_CANARY}"
    "${XE_CANARY}/third_party"
    "${XE_CANARY}/third_party/Vulkan-Headers/include")
# The platform switches ReXGlue sets for its builds (compat/rex/platform.h has them too).
target_compile_definitions(native-renderer PRIVATE
    XDROID360_NATIVE_RENDERER=1
    REX_PLATFORM_ANDROID=1 REX_PLATFORM_SWITCH=0 REX_PLATFORM_IOS=0 REX_PLATFORM_MAC=0
    REX_PLATFORM_WIN=0 REX_PLATFORM_LINUX=0 REX_ARCH_ARM=1 REX_ARCH_ARM64=1 REX_ARCH_X64=0
    VULKAN_HPP_NO_EXCEPTIONS XE_OPTION_PROFILING=0 XE_OPTION_PROFILING_UI=0)
target_compile_options(native-renderer PRIVATE -fexceptions -frtti)

# A compile check of the xenia-vulkan file the renderer needs changed (the device features).
option(NATIVE_RENDERER_CHECK_XENIA_DEVICE "Also compile xenia-vulkan's vulkan_device.cc" OFF)
if(NATIVE_RENDERER_CHECK_XENIA_DEVICE)
    add_library(native-renderer-check-device OBJECT
        "${XDROID360_SOURCE_ROOT}/xenia-vulkan/src/xenia/ui/vulkan/vulkan_device.cc")
    get_target_property(_includes native-renderer INCLUDE_DIRECTORIES)
    target_include_directories(native-renderer-check-device PRIVATE ${_includes})
    target_include_directories(native-renderer-check-device SYSTEM PRIVATE
        "${XE_CANARY}" "${XE_CANARY}/third_party" "${XE_CANARY}/third_party/Vulkan-Headers/include")
    target_compile_definitions(native-renderer-check-device PRIVATE
        VULKAN_HPP_NO_EXCEPTIONS XE_OPTION_PROFILING=0 XE_OPTION_PROFILING_UI=0)
endif()
